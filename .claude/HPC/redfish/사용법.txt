# 3. 통합 실행 bios.sh — 점검·전체비교·수정을 한 화면(TUI)에서 (기본 진입점, 재부팅은 하지 않음)
########################################################################

# (터미널) TUI: 대상 입력(파일 지정 / 붙여넣기: Ctrl+D 완료, Ctrl+X 취소) -> 프로파일 선택 -> 접속·조회(호스트당 로그인 1회)
# -> 결과 화면 [프로파일 점검] -> Tab: [전체 설정 비교] (처음이면 기준 선택 후 비교, 다시 Tab 이면 프로파일 점검으로)
# -> e -> 수정 단계 (벤더 줄에서 q 두 번은 묻지 않고 종료):
#    확인 1) 차이 항목 전부를 "현재값 -> 변경값 (수정대상)" / "(수정제외: 사유)" 로 보이고 "위 수정대상을 적용할까요?" y/n
#    확인 2) 수정제외 항목이 있으면 그것만 다시 보이고 "수정제외 항목을 확인했습니다. 진행할까요?" y/n
#    확인 3) 남의 Pending 이 있는 장비가 있으면 y 포함 / n 제외 / Ctrl+X 취소
#    확인 1·2 에서 n / Esc / Ctrl+X 이면 아무것도 쓰지 않고 결과 화면으로 돌아온다
#    e 를 누르면 먼저 "점검 결과 기준 수정" / "BIOS 값 직접 입력" 을 고른다
#    BIOS 값 직접 입력: 속성 이름(공백 또는 탭)설정값 을 한 줄에 하나씩 붙여 넣고 Ctrl+D (예: SysProfile PerfOptimized)
#      -> 이미 같은 값·없는 속성·보호 속성·맞지 않는 값인 대상만 건너뛰고 미리보기 -> y 면 나머지 설정
#    설정 뒤에는 설정을 보낸 호스트만 자동으로 다시 점검한다 (results/<일시>/recheck/<일시>/)
# 화면 키: Tab 화면 전환(어느 위치에서나) / <- -> 벤더 / Enter 벤더 안으로(<- -> 가 관리망 그룹) / r 전체 설정 비교 기준 다시 선택
#          q 어디서나 한 단계 뒤로 (벤더 줄에서는 q 를 한 번 더 눌러야 끝, 그때 Esc·N·Ctrl+X 는 종료 확인만 취소)
#          Esc 는 q 와 같이 뒤로 (벤더 줄에서는 종료 확인 취소) / 위아래(k j) 한 줄 / PgUp PgDn 한 화면
#          a 전체 o OK만 f difference만 n N/A만 / s 요약 / e 수정 단계로 바로 / ? 도움말 / Ctrl+X 진행 중 작업 취소
#          Ctrl+C·Ctrl+Z 는 무시(종료는 q 두 번)
# 수정 단계는 e 로만 간다 (q 두 번으로 끝낼 때는 수정 여부를 묻지 않는다). 거절한 뒤에도 e 로 다시 갈 수 있고,
# 설정 뒤 자동 재점검이 끝나면 같은 세션에서 다시 e 로 수정할 수 있다 (반영은 재부팅 후)
# 점검 끝 계정 생성: 2순위로 점검한 호스트 + user2 만 없는 호스트를 [Y/n] 으로 묻고,
#   1·2순위 계정이 모두 거부된 AUTH_FAIL 호스트는 목록을 보인 뒤 관리자 ID / 비밀번호(숨김)를 직접 입력해 없는 계정만 만든다
#   (그래도 실패한 호스트는 다시 나열, 빈 ID 입력이면 건너뜀, 최대 3번)
bash bios.sh

# 프로파일과 대상 파일을 미리 지정해 화면에서 묻지 않게 한다 (수정 단계의 확인 질문은 이때도 항상 묻는다)
bash bios.sh -profile VM -user user.txt

# 전체비교 기준까지 미리 지정하면 Tab 을 기다리지 않고 수집 직후 바로 전체 설정 비교까지 실행한다 (첫 화면은 [프로파일 점검])
bash bios.sh -profile VM -user user.txt -ref <비교서버 hostname>

# 연구용 프로파일 (전부 verified=N 이라 점검만 되고 설정되지 않음)
bash bios.sh -profile VM_research

# Cisco 장비는 XML API(읽기 전용)로 점검한다: hostname 이 bios.conf 의 xml_host_prefix(기본 c)로 시작하면 자동으로 XML,
# 그 밖의 hostname 과 IP 는 Redfish 다. 한 목록에 섞어 써도 된다. XML 대상은 점검만 하고 전체비교·설정에서는 빠진다
bash bios.sh -profile VM -user user.txt

# IP 형태의 Cisco 장비는 별도 파일에 적어 -user-xml 로 지정한다 (Redfish 대상은 -user, Cisco(XML) 대상은 -user-xml)
bash bios.sh -profile VM -user user.txt -user-xml user_xml.txt

# -user-xml 만 주면 user.txt 는 읽지 않고 그 목록만 XML 로 점검한다 (TUI 없이 평문 출력)
bash bios.sh -plain -profile VM -user-xml user_xml.txt

# 옛 방식(bios_check.sh 옛 옵션, all_bios_check.sh)도 같은 분류: c 로 시작하는 이름·-user-xml 대상은 XML 로 점검만 (설정·전체비교 제외)
bash bios_check.sh --profile VM -user user.txt -user-xml user_xml.txt -no-prompt

# 접두어는 bios.conf 에서 바꾼다 (쉼표로 여러 개, 비우면 이름으로는 XML 로 보내지 않고 -user-xml 목록만 XML)
grep -n xml_host_prefix bios.conf

# TUI 없이 색 없는 평문으로 출력한다 (읽기 전용, 설정 질문 없음. 파이프·파일로 저장할 때도 평문)
bash bios.sh -plain -profile VM

# 한 줄 형식: [OK]/[difference]/[N/A] host ip vendor model bios attribute 기준값 현재값 비고(원래 상태명). 색을 끄려면 NO_COLOR
NO_COLOR=1 bash bios.sh -plain -profile VM

# 접속 없이 xml.sh 로 저장한 덤프를 읽는다 (설정은 하지 않음)
bash bios.sh -plain -profile VM -from-dump dumps

# --- 전체비교 기준 3종 (같은 모델끼리만 비교, 제외 속성은 ignore_attrs.txt + bios.conf exclude_attrs) ---

# 기준 1: 비교서버 hostname (그 서버만 추가 조회, 쉼표·공백으로 여러 개). 평문 출력으로 BIOS 전체 값을 비교한다
bash bios.sh -plain -profile VM -ref <비교서버 hostname>

# 기준 1 을 TUI 에서: Tab -> [전체 설정 비교] 에서 기준 선택 "비교서버 hostname 입력" (비워 두면 diff.txt). -ref 를 주면 묻지 않고 바로 비교
bash bios.sh -profile VM -ref <비교서버 hostname>

# 기준 2: refs 폴더의 JSON (bios_export.sh 로 뽑은 속성 JSON). 파일 이름만 주면 bios.conf 의 refs_dir 에서 찾는다
bash bios.sh -plain -profile VM -ref-json <파일이름>.json

# 기준 3: diff.txt (모델별 기준 호스트 목록)
bash bios.sh -plain -profile VM -diff diff.txt

# 전체비교에서 제외할 속성 목록 파일을 직접 지정한다 (없으면 내장 기본 목록. bios.conf 의 exclude_attrs 는 항상 합쳐서 적용)
bash bios.sh -plain -profile VM -ref <비교서버 hostname> -ignore <제외속성 파일>

# --- 기준 JSON 을 refs/ 에 모으기 ---

# 비교서버의 BIOS 속성 JSON 을 뽑는다 (읽기 전용): results/<일시>_all/attrs/<호스트>.json
bash bios_export.sh <비교서버> <대상1> <대상2>

# refs 폴더에 복사해 두면 TUI 기준 선택 목록과 -ref-json 에 나온다 (실제 호스트 값이므로 git 에 올리지 말 것)
mkdir -p refs
cp results/<일시>_all/attrs/<비교서버>.json refs/

# --- auto_setup 비대화형 (점검만, 설정 없음, 묻지 않음) ---

# 호출한 폴더에 check.res_<user> 를 만든다. targets.txt 는 호출한 폴더 기준 경로. 프로파일은 -profile 또는 bios.conf 의 auto_profile
# 시작할 때 이전 check.res_<user>(와 _postapply)를 먼저 지우고, 새 파일은 임시 파일 + rename 으로 한 번에 만든다
bash bios.sh -auto-check <user> targets.txt -profile VM

# 위와 같고 check.res_<user>_postapply 도 같은 내용으로 만든다 (-auto 도 점검만 한다)
bash bios.sh -auto <user> targets.txt -profile VM

# 결과: 호스트마다 한 줄 (host: OK / host: OK (N/A N건) / host: FAIL difference N건 (...) / host: FAIL N/A 접속실패(...))
cat check.res_<user>

# FAIL 이 있어도 종료코드는 0 이다 (치명적 오류만 0 이 아님)
bash bios.sh -auto-check <user> targets.txt -profile VM; echo $?

# --- 옛 스크립트 호환 ---

# 옛 전용 옵션 없이 터미널에서 실행하면 bios.sh 로 넘어간다 (같은 TUI)
bash bios_check.sh --profile VM
bash all_bios_check.sh

# 옛 방식(Y/N 줄 프롬프트)은 -dry-run -no-prompt -stdin-ok -retry-from -list-max -fail-max 중 하나를 줄 때: 5~7절 참고
bash bios_check.sh --profile VM -no-prompt

# --- bios.conf 에서 쓰는 키 (bios.conf.example 참고, 줄 끝 주석 불가) ---

# 점검·전체비교에서 뺄 속성 패턴: | 구분, * 와일드카드, 대소문자 무시. 프로파일 점검에서는 해당 행을 점검하지 않는다
#   exclude_attrs=ServiceTag|*Uuid*|BootSeq*
# on: 전체비교 difference 도 "위 수정대상을 적용할까요?" 대상에 넣는다 / off(기본): 프로파일(verified=Y) 항목만 설정
# (on 이어도 비교서버와 같은 모델·같은 BIOS 버전·같은 속성 이름 목록이고, 적용 때 레지스트리 형식이 Enumeration·Integer·Boolean 인 속성만)
#   allcheck_set=off
