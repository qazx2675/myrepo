# [Step 0] 관리서버에서 ilo 브랜치 전용 저장소 클론 및 디렉토리 이동
git clone -b ilo --single-branch https://github.com/qazx2675/myrepo.git ilo-ip-changer && cd ilo-ip-changer

# [Step 0-1] 이미 클론된 디렉토리에서 최신 내용으로 업데이트할 때
git pull origin ilo

# [Step 1] 스크립트 실행 권한 부여 및 ilorest 정상 동작 확인
chmod +x change_ilo_ip.sh && /usr/sbin/ilorest --nologo --version

# [Step 1-1] RHEL 6 환경에서 ilorest "Cannot open self" 오류 발생 시 prelink 예외 등록 및 해제
echo "-b /usr/sbin/ilorest" >> /etc/prelink.conf && prelink -ua

# [Step 2] 샘플 목록 파일을 복사하여 작업용 list.txt 생성
cp list.txt.example list.txt

# [Step 2-1] 대상 서버 목록 편집 ([현재IP] [변경IP] [변경VLAN], VLAN 해제는 0 또는 off)
vim list.txt

# [Step 3] 실제 변경 전 전체 대상 장비 사전 점검 (로그인/포트/현재설정 조회만 수행)
bash change_ilo_ip.sh --check

# [Step 4] 대화형 일괄 변경 실행 (ID/PW/서브넷/게이트웨이/진행여부를 순서대로 질문)
bash change_ilo_ip.sh

# [Step 4-1] (선택) 서브넷·게이트웨이를 지정하고 확인 프롬프트 없이 일괄 변경할 때
ILO_USER="Administrator" bash change_ilo_ip.sh --netmask 255.255.254.0 --gw 10.20.30.254 -y

# [Step 5] 실행 결과 요약 로그 및 변경 전 원본 백업 JSON 확인
ls -lh logs/ && cat logs/result_*.log

# [Step 6] (검증용) Mock 기반 E2E 자동화 테스트 실행 (실제 장비에 접속하지 않음)
bash tests/run_tests.sh

# [부록] 작업 후 관리망 불통 시 대상 호스트 OS(root)에서 현재 iLO 네트워크 상태 확인
ipmitool lan print 1

# [부록] 대상 호스트 OS(root)에서 로컬 ilorest로 현재 iLO 포트 설정 확인
ilorest --nologo get --select EthernetInterface.

# [부록] 대상 호스트 OS(root)에서 ipmitool로 iLO IP/서브넷/게이트웨이/VLAN 비상 원복 예시
ipmitool lan set 1 ipsrc static && ipmitool lan set 1 ipaddr 10.10.10.101 && ipmitool lan set 1 netmask 255.255.255.0 && ipmitool lan set 1 defgw ipaddr 10.10.10.1 && ipmitool lan set 1 vlan id 30
