# auto_setup - 직접 칠 명령어 모음 (각 명령 위에 설명)
#
# 구성: os8_mgmt(RHEL8) 에만 데몬·상태(/tmp/auto_setup)가 있고, os6_mgmt(RHEL6) 에서는 같은 명령이 "원격 클라이언트 모드" 로
#       os8_mgmt 에 요청해서 어디서 실행해도 결과가 같다. 아래 [os8] 표시는 데몬이 있는 서버, [os6] 는 RHEL6 관리 서버.
# 주의: 소스·스크립트 최상단의 빈 변수(main.go: os6_mgmt os6_gossh os6_autosetup os_check_sh awx_dir os8_mgmt os8_autosetup os6_os_check_sh ldap_share_dir /
#       os_check: auto_done_dir auto_done_host / 01: auto_setup_host)는 "테스트 때만" 채우고 커밋하지 않는다 (7절 참고).

# ===== 1. 빌드 · 설치 [os8] =====

# 작업 폴더로 이동 (저장소 안의 auto_setup 폴더)
cd /root/saccae-src/.claude/HPC/auto_setup

# 빌드만 (현재 폴더에 ./auto_setup 생성, go 가 PATH 에 있어야 함)
go build -o auto_setup .

# 빌드 + /usr/local/bin 설치 + cron(매분 ensure) 등록 + tmpfiles 정리 제외 등록 (root, 여러 번 실행해도 안전)
bash setup.sh

# ===== 2. 상태 보기 · 데몬 제어 =====

# 현재 상태 리포트: 터미널이면 TUI(↑↓ 행, Enter 상세, ←→ 작업 전환, f 정체·실패만, c 수동 OS 체크, r 새로고침, ? 도움말, q 종료)
auto_setup

# TUI 가 아니라 텍스트 표로 보기 (파이프·로그용, 색 없음)
auto_setup --plain

# 색 없이 TUI 보기
NO_COLOR=1 auto_setup

# 데몬 기동 (이미 떠 있으면 "이미 실행 중" 안내만 — 보통은 cron 의 ensure 가 대신 기동)
auto_setup --start

# 데몬 재기동 (진행 중이던 작업은 상태 파일에서 이어서 복원됨)
auto_setup --restart

# 데몬 종료 (pid 파일 정리까지, cron 이 매분 다시 살리므로 완전히 막으려면 8절 '제거' 참고)
auto_setup --stop

# 데몬이 떠 있는지
pgrep -af 'auto_setup daemon'

# 작업별 진행 상태 한 줄 요약
auto_setup status

# 상태 스냅샷(JSON) 출력 (TUI·원격 클라이언트가 쓰는 것과 같은 내용)
auto_setup snapshot

# 완료 메시지(wall)에 나온 4자리 code 의 결과 코멘트 출력
auto_setup code 4821

# 무기한 감시 중인 작업 종료 (status 에 나온 jobid)
auto_setup cancel 20261002-153000-user1

# 다른 경로(수동 작업 등)로 이미 끝난 호스트를 완료 처리 — 부팅 시각 검증이 없으니 정말 끝난 호스트에만 (여러 대 가능)
auto_setup done web01 web02

# 수동 OS 체크 실행 요청: 해당 그룹(yml) 의 모든 호스트가 완료된 뒤에만 수락, 데몬 run 큐에서 실행됨 (TUI 의 c 키와 같음)
auto_setup request manual-run 20261002-153000-user1 gpu.yml

# 요청이 거부됐는지 확인 (마지막 줄 reason= 에 사유: 완료 전 호스트/job 없음 등)
cat /tmp/auto_setup/requests/rejected/*.req

# 로그 보기 (상태 변화·실행·오류만 기록됨, 비밀번호 값은 남지 않음)
tail -f /tmp/auto_setup/auto_setup.log

# 데몬을 수동으로 한 번 기동 (cron 이 하는 일과 같음, 중복 기동 안 됨)
auto_setup ensure

# ===== 3. 단계별 점검 (설명 → 실행 → 확인할 것 → Enter) =====

# 기본 점검 1~17단계 (데이터는 /tmp/auto_setup_manual 만 사용, 운영 /tmp/auto_setup 은 건드리지 않음)
bash test/manual_check.sh

# Enter 대기 없이 전 단계 연속 실행 (wall·설치 단계는 건너뜀, TUI 는 --plain 만 자동 확인)
bash test/manual_check.sh -y

# 11단계(상태 TUI 직접 확인)부터 다시 시작 — 1~10 은 1차 기능, 11~17 은 2차 기능
bash test/manual_check.sh -s 11

# 4단계 실제 ICMP 점검에 쓸 주소 지정 (응답하는 IP / 응답 없는 IP, root 필요)
PING_UP=192.168.0.59 PING_DOWN=192.168.0.250 bash test/manual_check.sh -s 4

# 9단계 설치 점검까지 포함 (setup.sh 실행 전에 yes 확인을 받음)
bash test/manual_check.sh --install

# ===== 4. 자동 테스트 =====

# Go 단위 테스트
go test ./...

# 정적 점검
go vet ./... && gofmt -l .

# 1차 목업 E2E (가짜 gossh/ssh/wall, 스크래치 사용, root 필요, 약 1분)
bash test/run_e2e.sh

# 2차 목업 E2E 전체: 양방향 동일성(a) 완료기록(b) 요청·TUI(c) 데몬 제어(d) LDAP(e) 2차 체크(f), root 필요, 약 3분
bash test/run_e2e2.sh

# 2차 E2E 를 시나리오 하나만 (예: 완료기록 b, LDAP e)
bash test/run_e2e2.sh b

# E2E 스크래치를 지우지 않고 남겨서 로그를 보고 싶을 때 (실행 끝에 경로가 출력됨)
KEEP_SCRATCH=1 bash test/run_e2e2.sh f

# ===== 5. os6_mgmt(RHEL6) 용 빌드 · 배치 =====

# [Go 1.20 서버(.60)] 정적 빌드 → auto_setup_os6 생성 (os6_mgmt 에서 ping 감시(probe) 하는 용도)
bash build_os6.sh

# [Go 1.20 서버] os6_mgmt 의 os6_autosetup 경로로 'auto_setup' 이름으로 복사 (경로·호스트는 환경에 맞게)
scp auto_setup_os6 root@<os6_mgmt>:<os6_autosetup 경로>/auto_setup

# [os6] 실행 권한 부여
ssh root@<os6_mgmt> "chmod 755 <os6_autosetup 경로>/auto_setup"

# 배치 확인 (stdin 목록을 ping 해서 'host up|down' 이 나오면 정상)
echo 127.0.0.1 | ssh root@<os6_mgmt> "<os6_autosetup 경로>/auto_setup probe"

# ===== 6. os6_mgmt 에서도 같은 명령을 쓰기 (원격 클라이언트 모드) =====

# [Go 1.20 서버] os6_mgmt 에서 쓸 "원격 클라이언트" 빌드: os8_mgmt 호스트와 그 서버의 auto_setup 경로를 빌드 때만 주입 (소스는 그대로)
CGO_ENABLED=0 GOOS=linux GOARCH=amd64 go build -ldflags "-X main.os8_mgmt=<os8_mgmt 호스트> -X main.os8_autosetup=<os8_mgmt 위 auto_setup 경로>" -o auto_setup_client .

# [Go 1.20 서버] os6_mgmt 로 복사 (이 파일이 os6_mgmt 의 auto_setup 명령이 됨 — 위 5절의 probe 용 빌드와는 다른 파일)
scp auto_setup_client root@<os6_mgmt>:/usr/local/bin/auto_setup

# [os6] 이후 os6_mgmt 에서도 2절의 명령이 똑같이 동작 (gossh 로 os8_mgmt 에 요청, 결과는 어디서 실행해도 동일)
auto_setup status

# [os6] 원격 클라이언트 빌드는 데몬 명령을 거부한다 (데몬은 os8_mgmt 에서만): "[X] 데몬은 os8_mgmt 에서만 실행합니다"
auto_setup --start

# 2차(이중) 체크를 쓰려면 [os8] 데몬 빌드에 os6_mgmt·os6_os_check_sh 를 같이 주입 (비우면 2차 체크 생략 = 1차와 동일)
go build -ldflags "-X main.os_check_sh=<이 서버의 os_check 전체경로> -X main.os6_mgmt=<os6_mgmt 호스트> -X main.os6_gossh=<os6_mgmt 의 gossh 전체경로> -X main.os6_autosetup=<os6_mgmt 의 auto_setup 디렉터리> -X main.os6_os_check_sh=<os6_mgmt 위 os_check 전체경로>" -o auto_setup .

# autofs 로 os8_mgmt·os6_mgmt 가 os_check·awx 스크립트를 같은 경로로 쓰는 경우: os6_os_check_sh 는 비워 두면 os_check_sh 와 같은 경로로 간주,
# awx_dir(awxkit/dhcp.sh 등이 있는 awx_script 디렉터리)를 채우면 dhcp.sh 를 그 경로에서 링크 (테스트·운영 빌드 때만 주입, 소스 커밋 금지)
go build -ldflags "-X main.os_check_sh=<autofs 공유 os_check 전체경로> -X main.awx_dir=<autofs 공유 awx_script 디렉터리> -X main.os6_mgmt=<os6_mgmt 호스트> -X main.os6_gossh=<os6_mgmt 의 gossh 전체경로> -X main.os6_autosetup=<os6_mgmt 의 auto_setup 디렉터리>" -o auto_setup .

# LDAP 복원(새 OS 의 binddn uid 가 백업과 다를 때)을 자동으로 하려면 ldap_share_dir(os8_mgmt·대상 서버·os6_mgmt 가 모두 보는 autofs 공유 경로)를 같이 주입.
# 비밀(bindpw)은 gossh 명령줄에 실리지 않고 이 공유경로에 0700/0600 으로 잠시 놓였다가 복원 직후 삭제된다. 대상 root 가 읽을 수 있어야 함(no_root_squash).
# 비워 두면 자동 복원은 하지 않고 "수동 복원 필요" 로 표시한다 (테스트·운영 빌드 때만 주입, 소스 커밋 금지)
go build -ldflags "-X main.os_check_sh=<os_check 전체경로> -X main.ldap_share_dir=<autofs 공유 디렉터리>" -o auto_setup .

# [os6] 01 을 os6_mgmt 에서 실행해도 전달이 os8_mgmt 에 되게 하려면: 01 복사본 최상단 auto_setup_host 를 채운다 (소스에는 커밋 금지)
sed -i 's|^auto_setup_host=""|auto_setup_host="<os8_mgmt 호스트>"|' 01.AWX_nodeinfo_V2.sh

# os_check 가 정상 종료 때 완료기록을 남기게 하려면 — os8_mgmt 위 os_check 복사본은 auto_done_dir(로컬 경로)
sed -i 's|^auto_done_dir=""|auto_done_dir="/tmp/auto_setup/done"|' os_check_final_annotated.sh

# os_check 가 정상 종료 때 완료기록을 남기게 하려면 — 다른 서버(os6_mgmt 등)의 os_check 복사본은 auto_done_host(os8_mgmt 호스트)
sed -i 's|^auto_done_host=""|auto_done_host="<os8_mgmt 호스트>"|' os_check_final_annotated.sh

# ===== 7. 빈 변수 채우기 (테스트 때만, 소스 수정·커밋 금지) =====

# 소스(main.go)는 건드리지 않고 빌드할 때만 값을 주입 (경로는 환경에 맞게)
go build -ldflags "-X main.os_check_sh=/절대경로/os_check_final_annotated.sh -X main.os6_mgmt=<os6_mgmt 호스트> -X main.os6_gossh=<os6_mgmt 의 gossh 전체경로> -X main.os6_autosetup=<os6_mgmt 의 auto_setup 디렉터리>" -o auto_setup .

# main.go 최상단 빈 변수 9개가 비어 있는지 확인 (커밋 전, 9줄 모두 = "" 여야 함)
grep -nE '^\s+(os6_mgmt|os6_gossh|os6_autosetup|os_check_sh|awx_dir|os8_mgmt|os8_autosetup|os6_os_check_sh|ldap_share_dir)\s+=' main.go

# os_check·01 의 새 빈 변수 3개도 커밋 전에 비어 있는지 확인 (3줄 모두 = "" 여야 함)
grep -nE '^(auto_done_dir|auto_done_host|auto_setup_host)=' "../OS 환경설정 체크/os_check_final_annotated.sh" ../awx_script/01.AWX_nodeinfo_V2.sh

# 실수로 main.go 를 고쳤다면 되돌리기
git checkout -- main.go

# ===== 8. 제거 =====

# 데몬 종료
pkill -f 'auto_setup daemon'

# cron 의 ensure 줄만 제거
crontab -l | grep -vF '/usr/local/bin/auto_setup ensure' | crontab -

# 설치 파일 · tmpfiles 설정 제거
rm -f /usr/local/bin/auto_setup /etc/tmpfiles.d/auto_setup.conf

# 데이터(작업 상태·code·로그·완료기록·LDAP 백업) 제거 — 진행 중 작업이 사라지므로 주의
rm -rf /tmp/auto_setup

# 점검용 데이터 제거 (manual_check.sh 를 중간에 끊었을 때)
pkill -f 'auto_setup daemon'; rm -rf /tmp/auto_setup_manual
